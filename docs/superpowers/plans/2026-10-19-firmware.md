# Display firmware Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Firmware for the Waveshare ESP32-S3 3.49" bar that shows the bridge's session cards (3 visible, swipe for more) and sends a `focus` action when a card is tapped.

**Architecture:** A pure C library `cards_core` (frame parser on jsmn, line assembler, age/status helpers, action formatter) is unit-tested on the host. ESP-IDF glue around it: `board` (display, touch, LVGL port, taken from Waveshare's `10_LVGL_V9_Test`), `serial_link` (USB-Serial-JTAG driver, rx task, action tx), `ui` (LVGL cards), `main`. One FreeRTOS queue (length 1, overwrite) carries parsed frames from the rx task to the UI.

**Tech Stack:** C11, ESP-IDF 5.5.3, LVGL 9.6 (managed component), jsmn v1.1.0 (vendored, MIT), host tests with plain `cc`.

## Global Constraints

- Wire format (from the bridge, one JSON line each way). Down: `{"v":1,"t":"state","now_ms":<int>,"total":<int>,"cards":[{"id":"<s>","title":"<s>","status":"working|waiting|idle|error|done","since_ms":<int>}]}`, at most 16 cards, id at most 64 bytes, title at most 48 runes, a frame at most 11992 bytes, newline-terminated; the bridge writes a leading `\n` after every reconnect. Up: `{"v":1,"t":"action","a":"focus","id":"<card id>"}\n`.
- Age is `now_ms - since_ms` plus the local milliseconds since the frame arrived. Never use a device clock.
- Frames arrive at most every 250 ms and at least every 30 s; "no bridge" shows after 35 s without a frame.
- The firmware ignores any line that is not a valid v1 state frame and never logs per bad line (counters only).
- Panel is 172x640 portrait, driven landscape 640x172 by the example's software rotation (`Rotated == USER_DISP_ROT_90`). Touch is I2C 0x3B. Keep the example's display/touch code as is.
- Target `esp32s3`, 16 MB flash, 8 MB octal PSRAM; app partition 8 MB.
- Stack limits: no large locals; frame and line buffers are static. The main task stack is 8192 bytes.
- Production C files under 200 lines (BSP copies excepted), functions under 50 lines.
- Hardware steps (flashing, running anything against the real board, tapping) are done by the controller with the user, never by subagents. Subagents must not run `idf.py flash`, `idf.py monitor`, `esptool`, the bridge without `--stdout`, or `cmd/boardcheck`.
- Commits: conventional style with the trailer given by the controller. No push.

```c
// Pure C core of the display firmware: no ESP-IDF dependencies, unit-tested on the host.
#ifndef CARDS_CORE_H
#define CARDS_CORE_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

#define CARDS_MAX 16
#define CARD_ID_MAX 65
#define CARD_TITLE_MAX 193
#define LINE_MAX_BYTES 12288

typedef enum {
	STATUS_WORKING,
	STATUS_WAITING,
	STATUS_IDLE,
	STATUS_ERROR,
	STATUS_DONE,
} card_status_t;

typedef struct {
	char id[CARD_ID_MAX];
	char title[CARD_TITLE_MAX];
	card_status_t status;
	int64_t since_ms;
} card_t;

typedef struct {
	int64_t now_ms;
	int total;
	int count;
	card_t cards[CARDS_MAX];
} frame_t;

typedef enum {
	FRAME_OK = 0,
	FRAME_BAD_JSON,
	FRAME_BAD_VERSION,
	FRAME_NOT_STATE,
	FRAME_BAD_FIELD,
	FRAME_TOO_MANY,
} frame_result_t;

// frame_parse parses one state frame line. out is written only on FRAME_OK.
// It uses static scratch memory and is not re-entrant: call it from one task.
frame_result_t frame_parse(const char *line, size_t len, frame_t *out);
const char *frame_result_name(frame_result_t r);

// linebuf assembles bytes into lines of at most LINE_MAX_BYTES-1 bytes.
typedef struct {
	char buf[LINE_MAX_BYTES];
	size_t len;
	bool overflow;
	bool ready;
} linebuf_t;

void linebuf_init(linebuf_t *lb);
// linebuf_push feeds one byte. It returns true when a complete non-empty line is in
// lb->buf (NUL-terminated, CR/LF stripped, *len bytes). Longer lines are dropped whole.
bool linebuf_push(linebuf_t *lb, char c, size_t *len);

// json_unescape decodes a JSON string body into UTF-8 in out (NUL-terminated). Text that
// does not fit is cut on a character boundary and *truncated is set. It returns false on
// an invalid escape. Malformed UTF-8 becomes U+FFFD.
bool json_unescape(const char *s, int len, char *out, size_t cap, bool *truncated);

void age_format(int64_t age_ms, char *out, size_t n);
int64_t card_age_ms(const frame_t *f, const card_t *c, int64_t since_rx_ms);
const char *status_label(card_status_t s);

// action_format writes the focus action line for a card id (with trailing newline) and
// returns its length, or -1 for an empty, overlong or unsafe id or a too small buffer.
int action_format(char *out, size_t n, const char *id);

#endif
```

```c
static const char GOOD[] =
	"{\"v\":1,\"t\":\"state\",\"now_ms\":1792399124168,\"total\":2,\"cards\":["
	"{\"id\":\"0000BBBB-0000-4000-8000-000000000004\",\"title\":\"DEV-24-delta\",\"status\":\"waiting\",\"since_ms\":1792317660000},"
	"{\"id\":\"0000BBBB-0000-4000-8000-000000000001\",\"title\":\"a\\u003cb\\\"c\",\"status\":\"working\",\"since_ms\":5}]}";

static frame_result_t parse(const char *s, frame_t *f) { return frame_parse(s, strlen(s), f); }

static void test_good_frame(void) {
	frame_t f;
	CHECK(parse(GOOD, &f) == FRAME_OK);
	CHECK(f.now_ms == 1792399124168LL);
	CHECK(f.total == 2 && f.count == 2);
	CHECK(strcmp(f.cards[0].id, "0000BBBB-0000-4000-8000-000000000004") == 0);
	CHECK(strcmp(f.cards[0].title, "DEV-24-delta") == 0);
	CHECK(f.cards[0].status == STATUS_WAITING);
	CHECK(f.cards[0].since_ms == 1792317660000LL);
	CHECK(strcmp(f.cards[1].title, "a<b\"c") == 0);
	CHECK(f.cards[1].status == STATUS_WORKING);
}

static void test_all_statuses_and_unknown(void) {
	frame_t f;
	const char *names[] = {"working", "waiting", "idle", "error", "done", "from-the-future"};
	card_status_t want[] = {STATUS_WORKING, STATUS_WAITING, STATUS_IDLE, STATUS_ERROR, STATUS_DONE, STATUS_IDLE};
	for (int i = 0; i < 6; i++) {
		char line[256];
		snprintf(line, sizeof line,
			"{\"v\":1,\"t\":\"state\",\"now_ms\":1,\"total\":1,\"cards\":[{\"id\":\"a\",\"title\":\"t\",\"status\":\"%s\",\"since_ms\":1}]}",
			names[i]);
		CHECK(parse(line, &f) == FRAME_OK);
		CHECK(f.cards[0].status == want[i]);
	}
}

static void test_empty_and_total(void) {
	frame_t f;
	CHECK(parse("{\"v\":1,\"t\":\"state\",\"now_ms\":9,\"total\":0,\"cards\":[]}", &f) == FRAME_OK);
	CHECK(f.count == 0 && f.total == 0 && f.now_ms == 9);
	// total smaller than the cards sent is raised to the count
	CHECK(parse("{\"v\":1,\"t\":\"state\",\"now_ms\":9,\"total\":0,\"cards\":[{\"id\":\"a\",\"title\":\"\",\"status\":\"idle\",\"since_ms\":1}]}", &f) == FRAME_OK);
	CHECK(f.total == 1);
	// total larger than the cards sent is kept (the cap was applied by the bridge)
	CHECK(parse("{\"v\":1,\"t\":\"state\",\"now_ms\":9,\"total\":40,\"cards\":[]}", &f) == FRAME_OK);
	CHECK(f.total == 40);
}

static void test_escapes_and_unicode(void) {
	frame_t f;
	CHECK(parse("{\"v\":1,\"t\":\"state\",\"now_ms\":1,\"total\":1,\"cards\":[{\"id\":\"a\",\"title\":\"\\u00e9\\ud83d\\ude00\\n\\/\",\"status\":\"idle\",\"since_ms\":1}]}", &f) == FRAME_OK);
	CHECK(strcmp(f.cards[0].title, "\xC3\xA9\xF0\x9F\x98\x80\n/") == 0);
	CHECK(parse("{\"v\":1,\"t\":\"state\",\"now_ms\":1,\"total\":1,\"cards\":[{\"id\":\"a\",\"title\":\"x\\ud83dy\",\"status\":\"idle\",\"since_ms\":1}]}", &f) == FRAME_OK);
	CHECK(strcmp(f.cards[0].title, "x\xEF\xBF\xBD" "y") == 0);
}

static void test_title_truncation(void) {
	frame_t f;
	char line[1024], title[400];
	memset(title, 'x', 300);
	title[300] = 0;
	snprintf(line, sizeof line, "{\"v\":1,\"t\":\"state\",\"now_ms\":1,\"total\":1,\"cards\":[{\"id\":\"a\",\"title\":\"%s\",\"status\":\"idle\",\"since_ms\":1}]}", title);
	CHECK(parse(line, &f) == FRAME_OK);
	CHECK(strlen(f.cards[0].title) == CARD_TITLE_MAX - 1);

	// a 2-byte character must not be cut in half
	char *p = title;
	*p++ = 'a';
	for (int i = 0; i < 100; i++) {
		*p++ = '\xC3';
		*p++ = '\xA9';
	}
	*p = 0;
	snprintf(line, sizeof line, "{\"v\":1,\"t\":\"state\",\"now_ms\":1,\"total\":1,\"cards\":[{\"id\":\"a\",\"title\":\"%s\",\"status\":\"idle\",\"since_ms\":1}]}", title);
	CHECK(parse(line, &f) == FRAME_OK);
	size_t n = strlen(f.cards[0].title);
	CHECK(n == 1 + 2 * 95);
	CHECK((unsigned char)f.cards[0].title[n - 2] == 0xC3 && (unsigned char)f.cards[0].title[n - 1] == 0xA9);
}

// The largest frame the bridge can send: 16 cards, 64-byte ids and 48-rune titles made of

static void test_worst_case_frame(void) {

	frame_t f;
	char id[64 * 6 + 1], title[48 * 6 + 1];
	id[0] = title[0] = 0;

	for (int i = 0; i < 48; i++) strcat(title, "\\u003c");

}
```

```c
#include <string.h>

#include "cards_core.h"
#include "test.h"

static linebuf_t lb;

// feed pushes s and collects completed lines separated by '|'.
static void feed(const char *s, char *out, size_t cap) {
	out[0] = 0;
	for (; *s; s++) {
		size_t len;
		if (linebuf_push(&lb, *s, &len)) {
			CHECK(strlen(lb.buf) == len);
			if (out[0]) strncat(out, "|", cap - strlen(out) - 1);
			strncat(out, lb.buf, cap - strlen(out) - 1);
		}
	}
}

static void test_lines(void) {
	char out[256];
	linebuf_init(&lb);
	feed("a\nbb\n", out, sizeof out);
	CHECK(strcmp(out, "a|bb") == 0);
	feed("\r\n\n\nc\r\n", out, sizeof out);
	CHECK(strcmp(out, "c") == 0);
}

static void test_partial_line_waits(void) {
	char out[256];
	linebuf_init(&lb);
	feed("{\"v\"", out, sizeof out);
	CHECK(out[0] == 0);
	feed(":1}\n", out, sizeof out);
	CHECK(strcmp(out, "{\"v\":1}") == 0);
}

static void test_resync_newline_prefix(void) {
	char out[256];
	linebuf_init(&lb);
	feed("half a li", out, sizeof out);
	feed("\n{\"ok\":1}\n", out, sizeof out);
	CHECK(strcmp(out, "half a li|{\"ok\":1}") == 0 || strcmp(out, "{\"ok\":1}") == 0);
}

static void test_overlong_line_is_dropped_whole(void) {
	linebuf_init(&lb);
	size_t len;
	for (int i = 0; i < LINE_MAX_BYTES + 500; i++) CHECK(!linebuf_push(&lb, 'x', &len));
	CHECK(!linebuf_push(&lb, '\n', &len));
	char out[64];
	feed("ok\n", out, sizeof out);
	CHECK(strcmp(out, "ok") == 0);
}

static void test_longest_accepted_line(void) {
	linebuf_init(&lb);
	size_t len = 0;
	bool done = false;
	for (int i = 0; i < LINE_MAX_BYTES - 1; i++) CHECK(!linebuf_push(&lb, 'y', &len));
	done = linebuf_push(&lb, '\n', &len);
	CHECK(done && len == LINE_MAX_BYTES - 1 && lb.buf[len] == 0);
}

void run_linebuf_tests(void) {
	test_lines();
	test_partial_line_waits();
	test_resync_newline_prefix();
	test_overlong_line_is_dropped_whole();
	test_longest_accepted_line();
}
```

```c
#include <stdio.h>
#include <string.h>

#include "cards_core.h"

static int hexval(char c) {
	if (c >= '0' && c <= '9') return c - '0';
	if (c >= 'a' && c <= 'f') return c - 'a' + 10;
	if (c >= 'A' && c <= 'F') return c - 'A' + 10;
	return -1;
}

static bool read_u4(const char *s, int len, int i, uint32_t *out) {
	if (i + 4 > len) return false;
	uint32_t v = 0;
	for (int k = 0; k < 4; k++) {
		int h = hexval(s[i + k]);
		if (h < 0) return false;
		v = (v << 4) | (uint32_t)h;
	}
	*out = v;
	return true;
}

static size_t put_utf8(uint32_t cp, char *d) {
	if (cp == 0) cp = 0xFFFD;
	if (cp < 0x80) {
		d[0] = (char)cp;
		return 1;
	}
	if (cp < 0x800) {
		d[0] = (char)(0xC0 | (cp >> 6));
		d[1] = (char)(0x80 | (cp & 0x3F));
		return 2;
	}
	if (cp < 0x10000) {
		d[0] = (char)(0xE0 | (cp >> 12));
		d[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
		d[2] = (char)(0x80 | (cp & 0x3F));
		return 3;
	}
	d[0] = (char)(0xF0 | (cp >> 18));
	d[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
	d[2] = (char)(0x80 | ((cp >> 6) & 0x3F));
	d[3] = (char)(0x80 | (cp & 0x3F));
	return 4;
}

static size_t seq_len(unsigned char lead) {
	if (lead < 0x80) return 1;
	if (lead >= 0xC2 && lead <= 0xDF) return 2;
	if (lead >= 0xE0 && lead <= 0xEF) return 3;
	if (lead >= 0xF0 && lead <= 0xF4) return 4;
	return 0;
}

// raw copies one UTF-8 sequence from s[*i..] into tmp, or U+FFFD for malformed bytes.
static size_t raw(const char *s, int len, int *i, char *tmp) {
	size_t l = seq_len((unsigned char)s[*i]);
	bool ok = l > 0 && *i + (int)l <= len;
	for (size_t k = 1; ok && k < l; k++) ok = ((unsigned char)s[*i + k] & 0xC0) == 0x80;
	if (!ok) {
		(*i)++;
		return put_utf8(0xFFFD, tmp);
	}
	memcpy(tmp, s + *i, l);
	*i += (int)l;
	return l;
}

static bool escape(const char *s, int len, int *i, uint32_t *cp) {
	if (*i + 1 >= len) return false;
	char e = s[*i + 1];
	*i += 2;
	switch (e) {
	case '"': *cp = '"'; return true;
	case '\\': *cp = '\\'; return true;
	case '/': *cp = '/'; return true;
	case 'b': *cp = '\b'; return true;
	case 'f': *cp = '\f'; return true;
	case 'n': *cp = '\n'; return true;
	case 'r': *cp = '\r'; return true;
	case 't': *cp = '\t'; return true;
	case 'u': break;
	default: return false;
	}
	if (!read_u4(s, len, *i, cp)) return false;
	*i += 4;
	if (*cp >= 0xD800 && *cp <= 0xDBFF) {
		uint32_t lo;
		if (*i + 6 <= len && s[*i] == '\\' && s[*i + 1] == 'u' && read_u4(s, len, *i + 2, &lo) &&
		    lo >= 0xDC00 && lo <= 0xDFFF) {
			*cp = 0x10000 + ((*cp - 0xD800) << 10) + (lo - 0xDC00);
			*i += 6;
		} else {
			*cp = 0xFFFD;
		}
	} else if (*cp >= 0xDC00 && *cp <= 0xDFFF) {
		*cp = 0xFFFD;
	}
	return true;
}

bool json_unescape(const char *s, int len, char *out, size_t cap, bool *truncated) {
	size_t o = 0;
	int i = 0;
	*truncated = false;
	if (cap == 0) return false;
	while (i < len) {
		char tmp[4];
		size_t n;
		if (s[i] != '\\') {
			n = raw(s, len, &i, tmp);
		} else {
			uint32_t cp;
			if (!escape(s, len, &i, &cp)) return false;
			n = put_utf8(cp, tmp);
		}
		if (o + n >= cap) {
			*truncated = true;
			break;
		}
		memcpy(out + o, tmp, n);
		o += n;
	}
	out[o] = '\0';
	return true;
}

void age_format(int64_t age_ms, char *out, size_t n) {
	long long s = age_ms < 0 ? 0 : (long long)(age_ms / 1000);
	if (s < 60) snprintf(out, n, "%llds", s);
	else if (s < 3600) snprintf(out, n, "%lldm", s / 60);
	else if (s < 86400) snprintf(out, n, "%lldh", s / 3600);
	else snprintf(out, n, "%lldd", s / 86400);
}

int64_t card_age_ms(const frame_t *f, const card_t *c, int64_t since_rx_ms) {
	if (since_rx_ms < 0) since_rx_ms = 0;
	int64_t a = f->now_ms - c->since_ms + since_rx_ms;
	return a < 0 ? 0 : a;
}

static bool safe_id_char(char c) {
	return (c >= '0' && c <= '9') || (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || c == '-' ||
	       c == '_' || c == '.' || c == ':';
```

```c
bool linebuf_push(linebuf_t *lb, char c, size_t *len) {
	if (lb->ready) {
		lb->len = 0;
		lb->ready = false;
	}
	if (c != '\n') {
		if (lb->overflow) return false;
		if (lb->len >= LINE_MAX_BYTES - 1) {
			lb->overflow = true;
			return false;
		}
		lb->buf[lb->len++] = c;
		return false;
	}
	bool dropped = lb->overflow;
	lb->overflow = false;
	if (lb->len > 0 && lb->buf[lb->len - 1] == '\r') lb->len--;
	if (dropped || lb->len == 0) {
		lb->len = 0;
		return false;
	}
	lb->buf[lb->len] = '\0';
	*len = lb->len;
	lb->ready = true;
	return true;
}
```

```c
#include <stdlib.h>
#include <string.h>

#define JSMN_STATIC
#define JSMN_STRICT
#include "jsmn.h"

#include "cards_core.h"

#define MAX_TOKENS 256

static jsmntok_t toks[MAX_TOKENS];
static frame_t scratch;

typedef struct {
	const char *js;
	int n;
} doc_t;

// skip returns the index after token i and all its descendants.
static int skip(const jsmntok_t *t, int i, int n) {
	int pending = 1;
	while (pending > 0 && i < n) {
		pending += t[i].size - 1;
		i++;
	}
	return i;
}

static bool str_is(const doc_t *d, const jsmntok_t *t, const char *s) {
	size_t len = strlen(s);
	return t->type == JSMN_STRING && (size_t)(t->end - t->start) == len && memcmp(d->js + t->start, s, len) == 0;
}

static bool to_i64(const doc_t *d, const jsmntok_t *t, int64_t *out) {
	if (t->type != JSMN_PRIMITIVE) return false;
	int n = t->end - t->start;
	if (n <= 0 || n > 20) return false;
	char b[24];
	memcpy(b, d->js + t->start, (size_t)n);
	b[n] = '\0';
	if (!(b[0] == '-' || (b[0] >= '0' && b[0] <= '9'))) return false;
	char *end;
	errno = 0;
	long long v = strtoll(b, &end, 10);
	if (errno || *end) return false;
	*out = v;
	return true;
}

static card_status_t to_status(const doc_t *d, const jsmntok_t *t) {
	if (str_is(d, t, "working")) return STATUS_WORKING;
	if (str_is(d, t, "waiting")) return STATUS_WAITING;
	if (str_is(d, t, "error")) return STATUS_ERROR;
	if (str_is(d, t, "done")) return STATUS_DONE;
	return STATUS_IDLE;
}

static frame_result_t read_card(const doc_t *d, const jsmntok_t *t, int i, card_t *c, int *next) {
	if (t[i].type != JSMN_OBJECT) return FRAME_BAD_FIELD;
	memset(c, 0, sizeof *c);
	c->status = STATUS_IDLE;
	bool has_id = false, has_since = false;
	int j = i + 1;
	for (int k = 0; k < t[i].size; k++) {
		if (j + 1 >= d->n || t[j].type != JSMN_STRING) return FRAME_BAD_JSON;
		const jsmntok_t *val = &t[j + 1];
		bool cut = false;
		if (str_is(d, &t[j], "id")) {
			if (val->type != JSMN_STRING ||
			    !json_unescape(d->js + val->start, val->end - val->start, c->id, sizeof c->id, &cut) || cut ||
			    c->id[0] == '\0')
				return FRAME_BAD_FIELD;
			has_id = true;
		} else if (str_is(d, &t[j], "title")) {
			if (val->type != JSMN_STRING ||
			    !json_unescape(d->js + val->start, val->end - val->start, c->title, sizeof c->title, &cut))
				return FRAME_BAD_FIELD;
		} else if (str_is(d, &t[j], "status")) {
			c->status = to_status(d, val);
		} else if (str_is(d, &t[j], "since_ms")) {
			if (!to_i64(d, val, &c->since_ms)) return FRAME_BAD_FIELD;
			has_since = true;
		}
		j = skip(t, j + 1, d->n);
	}
	*next = j;
	return (has_id && has_since) ? FRAME_OK : FRAME_BAD_FIELD;
}

static frame_result_t read_cards(const doc_t *d, const jsmntok_t *t, int i, frame_t *f) {
	if (t[i].type != JSMN_ARRAY) return FRAME_BAD_FIELD;
	if (t[i].size > CARDS_MAX) return FRAME_TOO_MANY;
	int j = i + 1;
	for (int m = 0; m < t[i].size; m++) {
		frame_result_t r = read_card(d, t, j, &f->cards[m], &j);
		if (r != FRAME_OK) return r;
	}
	f->count = t[i].size;
	return FRAME_OK;
}

typedef struct {
	bool v, t, now, cards;
	int64_t total;
} seen_t;

static frame_result_t read_top(const doc_t *d, const jsmntok_t *t, seen_t *s) {
	int j = 1;
	for (int k = 0; k < t[0].size; k++) {
		if (j + 1 >= d->n || t[j].type != JSMN_STRING) return FRAME_BAD_JSON;
		const jsmntok_t *val = &t[j + 1];
		int64_t v;
		if (str_is(d, &t[j], "v")) {
			if (!to_i64(d, val, &v) || v != 1) return FRAME_BAD_VERSION;
			s->v = true;
		} else if (str_is(d, &t[j], "t")) {
			if (!str_is(d, val, "state")) return FRAME_NOT_STATE;
			s->t = true;
		} else if (str_is(d, &t[j], "now_ms")) {
			if (!to_i64(d, val, &scratch.now_ms)) return FRAME_BAD_FIELD;
			s->now = true;
		} else if (str_is(d, &t[j], "total")) {
			if (!to_i64(d, val, &s->total) || s->total < 0 || s->total > 1000000) return FRAME_BAD_FIELD;
		} else if (str_is(d, &t[j], "cards")) {
			frame_result_t r = read_cards(d, t, j + 1, &scratch);
			if (r != FRAME_OK) return r;
			s->cards = true;
		}
		j = skip(t, j + 1, d->n);
	}
	return FRAME_OK;
}

frame_result_t frame_parse(const char *line, size_t len, frame_t *out) {
	if (len == 0 || len > LINE_MAX_BYTES) return FRAME_BAD_JSON;
	jsmn_parser p;
	jsmn_init(&p);
	int n = jsmn_parse(&p, line, len, toks, MAX_TOKENS);
	if (n < 1 || toks[0].type != JSMN_OBJECT) return FRAME_BAD_JSON;
	if (toks[0].end != (int)len) return FRAME_BAD_JSON;
	doc_t d = {.js = line, .n = n};
	memset(&scratch, 0, sizeof scratch);
	seen_t seen = {.total = -1};
	frame_result_t r = read_top(&d, toks, &seen);
	if (r != FRAME_OK) return r;
	if (!seen.v) return FRAME_BAD_VERSION;
	if (!seen.t) return FRAME_NOT_STATE;
	if (!seen.now || !seen.cards) return FRAME_BAD_FIELD;
	scratch.total = seen.total < scratch.count ? scratch.count : (int)seen.total;
	*out = scratch;
	return FRAME_OK;
}

const char *frame_result_name(frame_result_t r) {
	switch (r) {
	case FRAME_OK: return "ok";
	case FRAME_BAD_JSON: return "bad json";
	case FRAME_BAD_VERSION: return "bad version";
	case FRAME_NOT_STATE: return "not a state frame";
	case FRAME_BAD_FIELD: return "bad field";
	case FRAME_TOO_MANY: return "too many cards";
	}
	return "unknown";
}
```

```c
static void lvgl_flush_cb(lv_display_t *disp, const lv_area_t *area, uint8_t *color_p) {
	esp_lcd_panel_handle_t panel_handle = (esp_lcd_panel_handle_t)lv_display_get_user_data(disp);
	lv_draw_sw_rgb565_swap(color_p, lv_area_get_width(area) * lv_area_get_height(area));
	lv_display_rotation_t rotation = lv_display_get_rotation(disp);
	lv_area_t rotated_area;
	if (rotation != LV_DISPLAY_ROTATION_0) {
		lv_color_format_t cf = lv_display_get_color_format(disp);
		rotated_area = *area;
		lv_display_rotate_area(disp, &rotated_area);
		uint32_t src_stride = lv_draw_buf_width_to_stride(lv_area_get_width(area), cf);
		uint32_t dest_stride = lv_draw_buf_width_to_stride(lv_area_get_width(&rotated_area), cf);
		int32_t src_w = lv_area_get_width(area);
		int32_t src_h = lv_area_get_height(area);
		lv_draw_sw_rotate(color_p, lvgl_dest, src_w, src_h, src_stride, dest_stride, rotation, cf);
		area = &rotated_area;
	}

	const int flush_coun = (LVGL_SPIRAM_BUFF_LEN / LVGL_DMA_BUFF_LEN);
	const int offgap = (EXAMPLE_LCD_V_RES / flush_coun);
	const int dmalen = (LVGL_DMA_BUFF_LEN / 2);
	int offsetx1 = 0;
	int offsety1 = 0;
	int offsetx2 = EXAMPLE_LCD_H_RES;
	int offsety2 = offgap;

	uint16_t *map = (uint16_t *)lvgl_dest;
	xSemaphoreGive(flush_done_semaphore);
	for (int i = 0; i < flush_coun; i++) {
		xSemaphoreTake(flush_done_semaphore, portMAX_DELAY);
		memcpy(trans_buf_1, map, LVGL_DMA_BUFF_LEN);
		esp_lcd_panel_draw_bitmap(panel_handle, offsetx1, offsety1, offsetx2, offsety2, trans_buf_1);
		offsety1 += offgap;
		offsety2 += offgap;
		map += dmalen;
	}
	xSemaphoreTake(flush_done_semaphore, portMAX_DELAY);
	lv_disp_flush_ready(disp);
}

static void touch_read_cb(lv_indev_t *indev, lv_indev_data_t *indevData) {
	uint8_t read_touchpad_cmd[11] = {0xb5, 0xab, 0xa5, 0x5a, 0x0, 0x0, 0x0, 0x0e, 0x0, 0x0, 0x0};
	uint8_t buff[32] = {0};
	ESP_ERROR_CHECK_WITHOUT_ABORT(i2c_master_write_read_dev(disp_touch_dev_handle, read_touchpad_cmd, 11, buff, 32));
	uint16_t pointX = (((uint16_t)buff[2] & 0x0f) << 8) | (uint16_t)buff[3];
	uint16_t pointY = (((uint16_t)buff[4] & 0x0f) << 8) | (uint16_t)buff[5];
	if (buff[1] > 0 && buff[1] < 5) {
		indevData->state = LV_INDEV_STATE_PRESSED;
		if (pointX > EXAMPLE_LCD_H_RES) pointX = EXAMPLE_LCD_H_RES;
		if (pointY > EXAMPLE_LCD_V_RES) pointY = EXAMPLE_LCD_V_RES;
		indevData->point.x = (EXAMPLE_LCD_H_RES - pointX);
		indevData->point.y = (EXAMPLE_LCD_V_RES - pointY);
	} else {
		indevData->state = LV_INDEV_STATE_RELEASED;
	}
}

bool board_lock(int timeout_ms) {
	assert(lvgl_mux && "board_init must be called first");
	const TickType_t timeout_ticks = (timeout_ms == -1) ? portMAX_DELAY : pdMS_TO_TICKS(timeout_ms);
	return xSemaphoreTake(lvgl_mux, timeout_ticks) == pdTRUE;
}

void board_unlock(void) {
	assert(lvgl_mux && "board_init must be called first");
	xSemaphoreGive(lvgl_mux);
}

static void lvgl_port_task(void *arg) {
	uint32_t task_delay_ms = LVGL_TASK_MAX_DELAY_MS;
	for (;;) {
		if (board_lock(-1)) {
			task_delay_ms = lv_timer_handler();
			board_unlock();
		}
		if (task_delay_ms > LVGL_TASK_MAX_DELAY_MS) {
			task_delay_ms = LVGL_TASK_MAX_DELAY_MS;
		} else if (task_delay_ms < LVGL_TASK_MIN_DELAY_MS) {
			task_delay_ms = LVGL_TASK_MIN_DELAY_MS;
		}
		vTaskDelay(pdMS_TO_TICKS(task_delay_ms));
	}
}
```

```c
#include "serial_link.h"

#include "cards_core.h"
#include "driver/usb_serial_jtag.h"
#include "esp_err.h"

#define RX_CHUNK 128
#define WRITE_TIMEOUT_MS 50

static QueueHandle_t frame_q;
static volatile uint32_t n_frames, n_rejected, n_bytes;

static void on_line(const char *line, size_t len) {
	static frame_t f;
	if (frame_parse(line, len, &f) == FRAME_OK) {
		xQueueOverwrite(frame_q, &f);
		n_frames++;
	} else {
		n_rejected++;
	}
}

static void rx_task(void *arg) {
	static linebuf_t lb;
	uint8_t chunk[RX_CHUNK];
	linebuf_init(&lb);
	for (;;) {
		int n = usb_serial_jtag_read_bytes(chunk, sizeof chunk, pdMS_TO_TICKS(100));
		if (n <= 0) continue;
		n_bytes += (uint32_t)n;
		for (int i = 0; i < n; i++) {
			size_t len;
			if (linebuf_push(&lb, (char)chunk[i], &len)) on_line(lb.buf, len);
		}
	}
}

QueueHandle_t serial_link_start(void) {
	frame_q = xQueueCreate(1, sizeof(frame_t));
	assert(frame_q);
	usb_serial_jtag_driver_config_t cfg = USB_SERIAL_JTAG_DRIVER_CONFIG_DEFAULT();
	cfg.rx_buffer_size = 2048;
	cfg.tx_buffer_size = 512;
	ESP_ERROR_CHECK(usb_serial_jtag_driver_install(&cfg));
	xTaskCreatePinnedToCore(rx_task, "serial_rx", 4096, NULL, 3, NULL, 1);
	return frame_q;
}

serial_link_stats_t serial_link_stats(void) {
	serial_link_stats_t s = {.frames = n_frames, .rejected = n_rejected, .bytes = n_bytes};
	return s;
}

bool serial_send_focus(const char *id) {
	char buf[160];
	buf[0] = '\n'; // ends any log text the console left half-written, so the bridge sees a clean line
	int n = action_format(buf + 1, sizeof buf - 1, id);
	if (n < 0) return false;
	return usb_serial_jtag_write_bytes(buf, (size_t)n + 1, pdMS_TO_TICKS(WRITE_TIMEOUT_MS)) == n + 1;
}
```

```go
// Command boardcheck writes one state frame line to the display and prints what the display
// sends back. It opens the real serial port: run it by hand, never from tests.
package main

import (
	"bufio"
	"flag"
	"fmt"
	"io"
	"os"
	"strings"
	"time"

	"desktop-controller/bridge/internal/link"
)

func main() {
	frameFile := flag.String("frame-file", "", "file whose last line is sent as the state frame (e.g. output of bridge --stdout)")
	secs := flag.Int("secs", 5, "seconds to listen")
	vid := flag.String("vid", "303A", "USB vendor id")
	pid := flag.String("pid", "1001", "USB product id")
	flag.Parse()

	frame, err := lastLine(*frameFile)
	if err != nil {
		fmt.Fprintln(os.Stderr, "boardcheck:", err)
		os.Exit(1)
	}
	port, err := link.SerialOpener(*vid, *pid, 115200)()
	if err != nil {
		fmt.Fprintln(os.Stderr, "boardcheck:", err)
		os.Exit(1)
	}
	r, ok := port.(io.Reader)
	if !ok {
		fmt.Fprintln(os.Stderr, "boardcheck: port is not readable")
		os.Exit(1)
	}
	go func() {
		sc := bufio.NewScanner(r)
		sc.Buffer(make([]byte, 0, 64*1024), 64*1024)
		for sc.Scan() {
			line := sc.Text()
			if len(line) > 200 {
				line = line[:200] + "..."
			}
			fmt.Printf("< %s\n", line)
		}
	}()
	if frame != "" {
		if _, err := port.Write([]byte("\n" + frame + "\n")); err != nil {
			fmt.Fprintln(os.Stderr, "boardcheck: write:", err)
		} else {
			fmt.Printf("> sent %d bytes\n", len(frame)+2)
		}
	}
	time.Sleep(time.Duration(*secs) * time.Second)
	port.Close()
}

func lastLine(path string) (string, error) {
	if path == "" {
		return "", nil
	}
	b, err := os.ReadFile(path)
	if err != nil {
		return "", err
	}
	lines := strings.Split(strings.TrimSpace(string(b)), "\n")
	return strings.TrimSpace(lines[len(lines)-1]), nil
}
```

```c
	int i = (int)(intptr_t)lv_event_get_user_data(e);
	if (i < shown.count && tap_cb) tap_cb(shown.cards[i].id);
}

static lv_obj_t *make_label(lv_obj_t *parent, const lv_font_t *font, uint32_t color) {
	lv_obj_t *l = lv_label_create(parent);
	lv_obj_set_style_text_font(l, font, 0);
	lv_obj_set_style_text_color(l, lv_color_hex(color), 0);
	return l;
}

static void make_card(int i) {
	card_widget_t *c = &w[i];
	c->card = lv_obj_create(row);
	lv_obj_set_size(c->card, CARD_W, CARD_H);
	lv_obj_set_scrollable(c->card, false);
	lv_obj_set_style_radius(c->card, 12, 0);
	lv_obj_set_style_bg_color(c->card, lv_color_hex(COLOR_CARD), 0);
	lv_obj_set_style_bg_color(c->card, lv_color_hex(COLOR_CARD_PRESSED), LV_STATE_PRESSED);
	lv_obj_set_style_border_width(c->card, BAR_W, 0);
	lv_obj_set_style_border_side(c->card, LV_BORDER_SIDE_LEFT, 0);
	lv_obj_set_style_pad_all(c->card, CARD_PAD, 0);
	lv_obj_add_event_cb(c->card, card_clicked, LV_EVENT_CLICKED, (void *)(intptr_t)i);

	c->title = make_label(c->card, &lv_font_montserrat_16, COLOR_TEXT);
	lv_label_set_long_mode(c->title, LV_LABEL_LONG_MODE_DOTS);
	lv_obj_set_width(c->title, INNER_W);
	lv_obj_set_height(c->title, TITLE_LINES * lv_font_get_line_height(&lv_font_montserrat_16));
	lv_obj_align(c->title, LV_ALIGN_TOP_LEFT, 0, 0);

	c->status = make_label(c->card, &lv_font_montserrat_14, COLOR_DIM);
	lv_obj_align(c->status, LV_ALIGN_BOTTOM_LEFT, 0, 0);
	c->age = make_label(c->card, &lv_font_montserrat_14, COLOR_DIM);
	lv_obj_align(c->age, LV_ALIGN_BOTTOM_RIGHT, 0, 0);
	set_hidden(c->card, true);
}

static void fill_card(int i, const card_t *c, int64_t age_ms) {
	char age[16];
	age_format(age_ms, age, sizeof age);
	set_text(w[i].title, c->title);
	set_text(w[i].status, status_label(c->status));
	set_text(w[i].age, age);
	lv_obj_set_style_border_color(w[i].card, status_color(c->status), 0);
	lv_obj_set_style_text_color(w[i].status, status_color(c->status), 0);
	lv_obj_set_style_opa(w[i].card, c->status == STATUS_DONE ? LV_OPA_60 : LV_OPA_COVER, 0);
}

// refresh updates ages and the no-bridge / empty overlays from the stored frame.
static void refresh(void) {
	int64_t since_rx = rx_ms < 0 ? 0 : local_ms() - rx_ms;
	bool stale = rx_ms < 0 || since_rx > STALE_MS;
	set_text(overlay_label, rx_ms < 0 ? "waiting for bridge" : "no bridge");
	set_hidden(overlay, !stale);
	set_hidden(empty_label, stale || shown.count != 0);
	for (int i = 0; i < shown.count; i++) fill_card(i, &shown.cards[i], card_age_ms(&shown, &shown.cards[i], since_rx));
}

static void refresh_timer(lv_timer_t *t) {
	refresh();
}

static lv_obj_t *make_overlay(void) {

	overlay_label = make_label(o, &lv_font_montserrat_20, COLOR_DIM);

	return o;
}

void ui_init(ui_tap_cb_t on_tap) {
	tap_cb = on_tap;
	lv_obj_t *scr = lv_screen_active();
	lv_obj_set_style_bg_color(scr, lv_color_hex(COLOR_SCREEN), 0);
	lv_obj_set_style_bg_opa(scr, LV_OPA_COVER, 0);

	row = lv_obj_create(scr);
	lv_obj_set_size(row, SCREEN_W, SCREEN_H);
	lv_obj_set_pos(row, 0, 0);
	lv_obj_set_style_bg_opa(row, LV_OPA_TRANSP, 0);
	lv_obj_set_style_border_width(row, 0, 0);
	lv_obj_set_style_radius(row, 0, 0);
	lv_obj_set_style_pad_all(row, PAD, 0);
	lv_obj_set_style_pad_column(row, GAP, 0);
	lv_obj_set_flex_flow(row, LV_FLEX_FLOW_ROW);
	lv_obj_set_scroll_dir(row, LV_DIR_HOR);
	lv_obj_set_scroll_snap_x(row, LV_SCROLL_SNAP_START);
	lv_obj_set_scrollbar_mode(row, LV_SCROLLBAR_MODE_OFF);
	for (int i = 0; i < CARDS_MAX; i++) make_card(i);

	empty_label = make_label(scr, &lv_font_montserrat_20, COLOR_DIM);
	lv_label_set_text(empty_label, "no sessions");
	lv_obj_center(empty_label);
	set_hidden(empty_label, true);

	more_label = make_label(scr, &lv_font_montserrat_12, COLOR_DIM);
	lv_obj_align(more_label, LV_ALIGN_TOP_RIGHT, -4, 0);
	set_hidden(more_label, true);

	stats_label = make_label(scr, &lv_font_montserrat_12, COLOR_DIM);
	lv_obj_align(stats_label, LV_ALIGN_BOTTOM_RIGHT, -4, 0);

	overlay = make_overlay();
	lv_timer_create(refresh_timer, REFRESH_MS, NULL);
	refresh();
}

void ui_show_frame(const frame_t *f, int64_t rx_local_ms) {
	shown = *f;
	rx_ms = rx_local_ms;
	for (int i = 0; i < CARDS_MAX; i++) set_hidden(w[i].card, i >= f->count);
	if (f->total > f->count) {
		char b[16];
		snprintf(b, sizeof b, "+%d", f->total - f->count);
		set_text(more_label, b);
		set_hidden(more_label, false);
	} else {
		set_hidden(more_label, true);
	}
	refresh();
}

void ui_set_stats(uint32_t frames, uint32_t rejected) {
	char b[32];
	snprintf(b, sizeof b, "rx %u/%u", (unsigned)frames, (unsigned)rejected);
	set_text(stats_label, b);
}
```
