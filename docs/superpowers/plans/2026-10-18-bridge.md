# Bridge (cmux -> USB serial) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Go daemon that turns cmux's Claude Code hook events into session-card frames and writes them as JSON lines to the ESP32 over USB serial.

**Architecture:** `follower` tails `~/.cmuxterm/events.jsonl` and yields parsed `Event`s. A pure `state.Apply` folds events into per-surface cards. `protocol.Render` serialises the sorted cards to one JSON line. `link.Sender` writes the latest line to the board, reconnecting when the port disappears.

**Tech Stack:** Go 1.27 (stdlib only, plus `go.bug.st/serial` for port discovery/IO), stdlib `testing`.

## Global Constraints

- Transport: USB CDC serial, newline-delimited JSON, protocol version `1`.
- Board USB ID: VID `303A`, PID `1001` (Espressif USB-Serial/JTAG); port is currently `/dev/cu.usbmodem101` but must be discovered by VID/PID, and re-discovered after reset.
- Event source for v1 is the file `~/.cmuxterm/events.jsonl` (the cmux socket is denied outside a cmux terminal).
- Status comes from cmux's own `sidebar.metadata.updated` events with `command":"set_status"` and args `claude_code <Label> ... --tab=<workspace> --panel=<surface>` (top-level `surface_id` is null on these). Labels seen in real data: `Running`, `Running subagents`, `Waiting`, `Needs input`, `Idle`, `API error`, `Connection dropped`. `Waiting` (hourglass) is treated as working; confirm on the display.
- Hook `Notification` is NOT a status signal (it fires 60 s after `Stop` as an idle ping).
- One card per cmux surface. Title = basename of the `cwd` from that surface's hook events, falling back to the first 8 characters of the surface id.
- Cards with no status change for `--max-age` (12h) are dropped, whatever their status (covers sessions killed without a Stop).
- Frames are emitted at most every 250 ms (the startup replay is ~16k events).
- Opening the serial port sets DTR and RTS low so the ESP32 is not reset on connect.
- `bridge/` needs a git repo for the checkpoint steps: ask the user before `git init`.
- No mutation of shared state: `Apply` returns a new `State`.
- Files under 200 lines, functions under 50 lines, errors never swallowed.
- Frames are always the full state (spec's `patch` frames and `cmux sessions --json` snapshot are deferred; frames are capped at 16 cards and 48-rune titles; worst case is ~12 KB, so the firmware line buffer must be sized for that).
- Do not commit without the user's go-ahead (CLAUDE.md). The "Commit" steps are checkpoints: ask first.

```go
// Package protocol defines the JSON-lines frames sent to the display.
package protocol

import (
	"encoding/json"
	"time"
)

const Version = 1

type Status string

const (
	Working Status = "working"
	Waiting Status = "waiting"
	Idle    Status = "idle"
	Error   Status = "error"
)

type Card struct {
	ID      string `json:"id"`
	Title   string `json:"title"`
	Status  Status `json:"status"`
	SinceMs int64  `json:"since_ms"`
}

type frame struct {
	V     int    `json:"v"`
	T     string `json:"t"`
	NowMs int64  `json:"now_ms"`
	Cards []Card `json:"cards"`
}

// Render returns the state frame as a single newline-terminated JSON line.
func Render(cards []Card, now time.Time) ([]byte, error) {
	if cards == nil {
		cards = []Card{}
	}
	b, err := json.Marshal(frame{V: Version, T: "state", NowMs: now.UnixMilli(), Cards: cards})
	if err != nil {
		return nil, err
	}
	return append(b, '\n'), nil
}
```

```go
// Package events parses cmux's newline-delimited event stream.
package events

import (
	"encoding/json"
	"fmt"
	"strings"
	"time"
)

const (
	statusKey   = "claude_code "
	statusEvent = "sidebar.metadata.updated"
)

type Event struct {
	Seq         int64
	Name        string
	At          time.Time
	WorkspaceID string
	SurfaceID   string
	Cwd         string
	Hook        string
	Label       string
}

type wire struct {
	Seq         int64     `json:"seq"`
	Name        string    `json:"name"`
	At          time.Time `json:"occurred_at"`
	WorkspaceID *string   `json:"workspace_id"`
	SurfaceID   *string   `json:"surface_id"`
	Payload     struct {
		Hook    string `json:"hook_event_name"`
		Cwd     string `json:"cwd"`
		Command string `json:"command"`
		Args    string `json:"args"`
	} `json:"payload"`
}

func deref(s *string) string {
	if s == nil {
		return ""
	}
	return *s
}

func ParseLine(line []byte) (Event, error) {
	var w wire
	if err := json.Unmarshal(line, &w); err != nil {
		return Event{}, fmt.Errorf("parse event: %w", err)
	}
	e := Event{
		Seq:         w.Seq,
		Name:        w.Name,
		At:          w.At,
		WorkspaceID: deref(w.WorkspaceID),
		SurfaceID:   deref(w.SurfaceID),
		Cwd:         w.Payload.Cwd,
		Hook:        w.Payload.Hook,
	}
	if w.Name == statusEvent && w.Payload.Command == "set_status" {
		e = withStatus(e, w.Payload.Args)
	}
	return e, nil
}

// withStatus reads "claude_code <Label> --flag=v ..." into Label, SurfaceID and WorkspaceID.
func withStatus(e Event, args string) Event {
	rest, ok := strings.CutPrefix(args, statusKey)
	if !ok {
		return e
	}
	label, flags, _ := strings.Cut(rest, " --")
	e.Label = strings.TrimSpace(label)
	for _, f := range strings.Fields(flags) {
		f = strings.TrimLeft(f, "-")
		if v, ok := strings.CutPrefix(f, "panel="); ok {
			e.SurfaceID = v
		}
		if v, ok := strings.CutPrefix(f, "tab="); ok {
			e.WorkspaceID = v
		}
	}
	return e
}
```

Rules:
- Events with an empty `SurfaceID` are ignored.
- A hook event with a `Cwd` records the surface's title (`filepath.Base(cwd)`); it never changes status.
- A status event (`Label` set) maps to a status: `Running`, `Running subagents`, `Waiting` -> Working; `Needs input` -> Waiting; `Idle` -> Idle; `API error`, `Connection dropped` -> Error. Unknown labels are ignored.
- `SinceMs` changes only when the status changes.
- A card's title is its recorded title, or the first 8 characters of the surface id when none is known (resolved when `Cards()` is called, so a late hook still fixes the title).
- `Prune` drops every card whose `SinceMs` is older than `maxAge`, whatever its status.

```go
// Package state folds cmux events into display cards.
package state

import (
	"path/filepath"
	"sort"
	"time"

	"desktop-controller/bridge/internal/events"
	"desktop-controller/bridge/internal/protocol"
)

const fallbackTitleLen = 8

// State is immutable: Apply and Prune return new values and never edit the maps they were given.
type State struct {
	cards  map[string]protocol.Card
	titles map[string]string
}

var labelStatus = map[string]protocol.Status{
	"Running":            protocol.Working,
	"Running subagents":  protocol.Working,
	"Waiting":            protocol.Working,
	"Needs input":        protocol.Waiting,
	"Idle":               protocol.Idle,
	"API error":          protocol.Error,
	"Connection dropped": protocol.Error,
}

func withTitle(s State, id, title string) State {
	if s.titles[id] == title {
		return s
	}
	titles := make(map[string]string, len(s.titles)+1)
	for k, v := range s.titles {
		titles[k] = v
	}
	titles[id] = title
	return State{cards: s.cards, titles: titles}
}

func withCard(s State, c protocol.Card) State {
	cards := make(map[string]protocol.Card, len(s.cards)+1)
	for k, v := range s.cards {
		cards[k] = v
	}
	cards[c.ID] = c
	return State{cards: cards, titles: s.titles}
}

func Apply(s State, e events.Event) State {
	if e.SurfaceID == "" {
		return s
	}
	if e.Hook != "" && e.Cwd != "" {
		s = withTitle(s, e.SurfaceID, filepath.Base(e.Cwd))
	}
	status, ok := labelStatus[e.Label]
	if !ok {
		return s
	}
	if card, seen := s.cards[e.SurfaceID]; seen && card.Status == status {
		return s
	}
	return withCard(s, protocol.Card{ID: e.SurfaceID, Status: status, SinceMs: e.At.UnixMilli()})
}

// Prune drops cards whose last status change is older than maxAge.
func Prune(s State, now time.Time, maxAge time.Duration) State {
	cutoff := now.Add(-maxAge).UnixMilli()
	cards := make(map[string]protocol.Card, len(s.cards))
	for k, c := range s.cards {
		if c.SinceMs >= cutoff {
			cards[k] = c
		}
	}
	return State{cards: cards, titles: s.titles}
}

var rank = map[protocol.Status]int{protocol.Waiting: 0, protocol.Error: 1, protocol.Working: 2, protocol.Idle: 3}

func (s State) title(id string) string {
	if t, ok := s.titles[id]; ok {
		return t
	}
	if len(id) > fallbackTitleLen {
		return id[:fallbackTitleLen]
	}
	return id
}

func (s State) Cards() []protocol.Card {
	out := make([]protocol.Card, 0, len(s.cards))
	for id, c := range s.cards {
		c.Title = s.title(id)
		out = append(out, c)
	}
	sort.Slice(out, func(i, j int) bool {
		if rank[out[i].Status] != rank[out[j].Status] {
			return rank[out[i].Status] < rank[out[j].Status]
		}
		if out[i].SinceMs != out[j].SinceMs {
			return out[i].SinceMs > out[j].SinceMs
		}
		return out[i].ID < out[j].ID
	})
	return out
}
```

```go
// Package follow tails a cmux events file, replaying history first.
package follow

import (
	"bufio"
	"bytes"
	"context"
	"io"
	"log"
	"os"
	"time"

	"desktop-controller/bridge/internal/events"
)

type Event = events.Event

// Follow replays path, then emits appended events until ctx is cancelled.
func Follow(ctx context.Context, path string, poll time.Duration) <-chan Event {
	out := make(chan Event, 64)
	go func() {
		defer close(out)
		var offset int64
		var partial []byte
		ticker := time.NewTicker(poll)
		defer ticker.Stop()
		for {
			offset, partial = drain(ctx, path, offset, partial, out)
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
			}
		}
	}()
	return out
}

func drain(ctx context.Context, path string, offset int64, partial []byte, out chan<- Event) (int64, []byte) {
	f, err := os.Open(path)
	if err != nil {
		if !os.IsNotExist(err) {
			log.Printf("follow: open %s: %v", path, err)
		}
		return offset, partial
	}
	defer f.Close()
	info, err := f.Stat()
	if err != nil {
		log.Printf("follow: stat %s: %v", path, err)
		return offset, partial
	}
	if info.Size() < offset {
		offset, partial = 0, nil
	}
	if _, err := f.Seek(offset, io.SeekStart); err != nil {
		log.Printf("follow: seek %s: %v", path, err)
		return offset, partial
	}
	r := bufio.NewReader(f)
	for {
		chunk, err := r.ReadBytes('\n')
		offset += int64(len(chunk))
		partial = append(partial, chunk...)
		if err != nil {
			return offset, partial
		}
		if e, perr := events.ParseLine(bytes.TrimSpace(partial)); perr == nil {
			select {
			case out <- e:
			case <-ctx.Done():
				return offset, nil
			}
		}
		partial = nil
	}
}
```

```go
// Package link writes frames to the display over USB serial.
package link

import (
	"context"
	"fmt"
	"io"
	"log"
	"strings"
	"time"

	"go.bug.st/serial"
	"go.bug.st/serial/enumerator"
)

type OpenFunc func() (io.WriteCloser, error)

type Sender struct {
	open  OpenFunc
	retry time.Duration
}

func NewSender(open OpenFunc, retry time.Duration) *Sender {
	return &Sender{open: open, retry: retry}
}

// Run writes the latest frame whenever one arrives and after every reconnect.
func (s *Sender) Run(ctx context.Context, frames <-chan []byte) {
	var latest []byte
	var port io.WriteCloser
	defer func() {
		if port != nil {
			port.Close()
		}
	}()
	for {
		if port == nil {
			p, err := s.open()
			if err != nil {
				log.Printf("link: open: %v", err)
				if !sleep(ctx, s.retry) {
					return
				}
				latest = drainLatest(frames, latest)
				continue
			}
			port = p
			if latest != nil && !s.write(&port, latest) {
				continue
			}
		}
		select {
		case <-ctx.Done():
			return
		case f, ok := <-frames:
			if !ok {
				return
			}
			latest = f
			s.write(&port, latest)
		}
	}
}

func (s *Sender) write(port *io.WriteCloser, frame []byte) bool {
	if _, err := (*port).Write(frame); err != nil {
		log.Printf("link: write: %v", err)
		(*port).Close()
		*port = nil
		return false
	}
	return true
}

func drainLatest(frames <-chan []byte, latest []byte) []byte {
	for {
		select {
		case f := <-frames:
			latest = f
		default:
			return latest
		}
	}
}

func sleep(ctx context.Context, d time.Duration) bool {
	select {
	case <-ctx.Done():
		return false
	case <-time.After(d):
		return true
	}
}

// openQuiet opens the port with DTR and RTS low so the ESP32 is not reset on connect.
func openQuiet(name string, baud int) (io.WriteCloser, error) {
	p, err := serial.Open(name, &serial.Mode{BaudRate: baud})
	if err != nil {
		return nil, fmt.Errorf("open %s: %w", name, err)
	}
	if err := p.SetDTR(false); err != nil {
		p.Close()
		return nil, fmt.Errorf("clear DTR: %w", err)
	}
	if err := p.SetRTS(false); err != nil {
		p.Close()
		return nil, fmt.Errorf("clear RTS: %w", err)
	}
	return p, nil
}

// SerialOpener finds the port by USB VID/PID on every call, so it survives resets.
func SerialOpener(vid, pid string, baud int) OpenFunc {
	return func() (io.WriteCloser, error) {
		ports, err := enumerator.GetDetailedPortsList()
		if err != nil {
			return nil, fmt.Errorf("list ports: %w", err)
		}
		for _, p := range ports {
			if p.IsUSB && strings.EqualFold(p.VID, vid) && strings.EqualFold(p.PID, pid) {
				return openQuiet(p.Name, baud)
			}
		}
		return nil, fmt.Errorf("no USB device %s:%s", vid, pid)
	}
}
```

```go
package main

import (
	"context"
	"flag"
	"io"
	"log"
	"os"
	"os/signal"
	"path/filepath"
	"time"

	"desktop-controller/bridge/internal/follow"
	"desktop-controller/bridge/internal/link"
	"desktop-controller/bridge/internal/protocol"
	"desktop-controller/bridge/internal/state"
)

const (
	emitEvery     = 250 * time.Millisecond
	heartbeatEvery = 30 * time.Second
)

type nopCloser struct{ io.Writer }

func (nopCloser) Close() error { return nil }

func main() {
	home, err := os.UserHomeDir()
	if err != nil {
		log.Fatal(err)
	}
	eventsPath := flag.String("events", filepath.Join(home, ".cmuxterm", "events.jsonl"), "cmux events file")
	vid := flag.String("vid", "303A", "USB vendor id")
	pid := flag.String("pid", "1001", "USB product id")
	baud := flag.Int("baud", 115200, "serial baud rate")
	toStdout := flag.Bool("stdout", false, "print frames instead of using serial")
	maxAge := flag.Duration("max-age", 12*time.Hour, "drop cards with no status change for this long")
	flag.Parse()

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
	defer stop()

	open := link.SerialOpener(*vid, *pid, *baud)
	if *toStdout {
		open = func() (io.WriteCloser, error) { return nopCloser{os.Stdout}, nil }
	}
	frames := make(chan []byte, 1)
	go link.NewSender(open, time.Second).Run(ctx, frames)

	run(ctx, follow.Follow(ctx, *eventsPath, 250*time.Millisecond), frames, *maxAge)
}

func run(ctx context.Context, evs <-chan follow.Event, frames chan []byte, maxAge time.Duration) {
	var s state.State
	dirty := true
	lastSent := time.Time{}
	tick := time.NewTicker(emitEvery)
	defer tick.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case e, ok := <-evs:
			if !ok {
				return
			}
			s = state.Apply(s, e)
			dirty = true
		case now := <-tick.C:
			if !dirty && now.Sub(lastSent) < heartbeatEvery {
				continue
			}
			s = state.Prune(s, now, maxAge)
			b, err := protocol.Render(s.Cards(), now)
			if err != nil {
				log.Printf("render: %v", err)
				continue
			}
			select {
			case <-frames:
			default:
			}
			frames <- b
			dirty, lastSent = false, now
		}
	}
}
```
